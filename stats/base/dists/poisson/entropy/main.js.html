'use strict';
 
// MODULES //
 
var isnan = require( '@stdlib/math/base/assert/is-nan' );
var floor = require( '@stdlib/math/base/special/floor' );
var log1p = require( '@stdlib/math/base/special/log1p' );
var exp = require( '@stdlib/math/base/special/exp' );
var ln = require( '@stdlib/math/base/special/ln' );
var LN_TWO_PI = require( '@stdlib/constants/float64/ln-two-pi' );
var EPS = require( '@stdlib/constants/float64/eps' );
 
 
// VARIABLES //
 
// Value of `lambda` above which the asymptotic expansion is accurate to within double-precision (relative error < 1e-16):
var THRESHOLD = 5000.0;
 
// Relative weight below which the probability mass of a term no longer affects the sum:
var TOL = EPS * 1.0e-3;
 
 
// FUNCTIONS //
 
/**
* Returns `ln(a/b)` for positive `a` and `b`.
*
* ## Notes
*
* -   When `a` and `b` are within a factor of two of each other, `a-b` is exact (Sterbenz lemma), and `log1p` avoids the cancellation which `ln(a/b)` suffers when `a/b` is close to one.
*
* @private
* @param {PositiveNumber} a - numerator
* @param {PositiveNumber} b - denominator
* @returns {number} natural logarithm of the ratio
*/
function lnratio( a, b ) {
	var r = a / b;
	if ( r > 0.5 && r < 2.0 ) {
		return log1p( ( a-b ) / b );
	}
	return ln( r );
}
 
 
// MAIN //
 
/**
* Returns the entropy of a Poisson distribution.
*
* ## Method
*
* For \\(\lambda < 5000\\), the function evaluates the definition \\(H = -\sum_k p_k \ln p_k\\) using weights relative to the mode \\(m = \lfloor \lambda \rfloor\\), \\(w_k = p_k / p_m\\). As the probabilities sum to one, \\(p_m = 1/W\\), where \\(W = \sum_k w_k\\), and thus
*
* ```tex
* H = \ln W - \frac{1}{W} \sum_k w_k \ln w_k
* ```
*
* The logarithms \\(\ln w_k\\) are accumulated using the recurrences \\(w_{k+1} = w_k \lambda/(k+1)\\) and \\(w_{k-1} = w_k k/\lambda\\). This avoids evaluating \\(\lambda^k/k!\\), which overflows for large \\(k\\), and avoids the cancellation between \\(k \ln \lambda\\), \\(\lambda\\), and \\(\ln k!\\), each of which is much larger than the entropy.
*
* For \\(\lambda \geq 5000\\), the function uses the asymptotic expansion
*
* ```tex
* H = \frac{1}{2} \ln(2 \pi e \lambda) - \frac{1}{12\lambda} - \frac{1}{24\lambda^2} - \frac{19}{360\lambda^3} + O\left(\lambda^{-4}\right)
* ```
*
* whose truncation error is less than \\(10^{-16}\\) relative to the result.
*
* @param {NonNegativeNumber} lambda - mean parameter
* @returns {PositiveNumber} entropy
*
* @example
* var v = entropy( 9.0 );
* // returns ~2.508
*
* @example
* var v = entropy( 1.0 );
* // returns ~1.305
*
* @example
* var v = entropy( -0.2 );
* // returns NaN
*
* @example
* var v = entropy( NaN );
* // returns NaN
*/
function entropy( lambda ) {
	var sum;
	var wm1;
	var lw;
	var l2;
	var w;
	var m;
	var k;
	if ( isnan( lambda ) || lambda < 0.0 ) {
		return NaN;
	}
	if ( lambda === 0.0 ) {
		return 0.0;
	}
	if ( lambda >= THRESHOLD ) {
		l2 = lambda * lambda;
		return ( 0.5 * ( LN_TWO_PI + 1.0 + ln( lambda ) ) ) - ( 1.0 / ( 12.0*lambda ) ) - ( 1.0 / ( 24.0*l2 ) ) - ( 19.0 / ( 360.0*l2*lambda ) ); // eslint-disable-line max-len
	}
	m = floor( lambda );
	wm1 = 0.0; // W - 1
	sum = 0.0; // Σ w_k ln(w_k)
 
	// Terms above the mode, using w_{k+1} = w_k * λ/(k+1):
	lw = 0.0;
	for ( k = m; ; k++ ) {
		lw += lnratio( lambda, k+1 );
		w = exp( lw );
		wm1 += w;
		sum += w * lw;
		if ( w < TOL ) {
			break;
		}
	}
	// Terms below the mode, using w_{k-1} = w_k * k/λ:
	lw = 0.0;
	for ( k = m; k > 0; k-- ) {
		lw += lnratio( k, lambda );
		w = exp( lw );
		wm1 += w;
		sum += w * lw;
		if ( w < TOL ) {
			break;
		}
	}
	return log1p( wm1 ) - ( sum / ( 1.0+wm1 ) );
}
 
 
// EXPORTS //
 
module.exports = entropy;
