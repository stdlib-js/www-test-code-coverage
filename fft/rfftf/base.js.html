/**
* Computes the forward FFT and writes results to the output complex ndarray.
*
* @private
* @param {Function} reinterpret - complex-to-real reinterpret function
* @param {Function} rffti - workspace initialization function
* @param {Function} rfftf - forward FFT function
* @param {Function} Workspace - typed array constructor for workspace
* @param {ArrayLikeObject<Object>} arrays - array-like object containing ndarrays
* @returns {void}
*/
function kernel( reinterpret, rffti, rfftf, Workspace, arrays ) {
	var view;
	var out;
	var M;
	var N;
	var w;
	var x;
 
	out = arrays[ 0 ];
	x = arrays[ 1 ];
	N = numelDimension( x, 0 );
	M = floor( N/2 ) + 1;
	view = reinterpret( getData( out ), 0 );
 
	if ( N === 0 ) {
		return;
	}
	// Copy the input into the output buffer at offset 1:
	gcopy( N, getData( x ), getStride( x, 0 ), getOffset( x ), view, 1, 1 );
 
	// Initialize the workspace and compute the forward FFT in-place:
	w = new Workspace( ( 2*N ) + 34 );
	rffti( N, w, 1, 0 );
	rfftf( N, view, 1, 1, w, 1, 0 );
 
	// Move the DC component from the imaginary slot to the real slot of the first complex element and zero out the imaginary part:
	view[ 0 ] = view[ 1 ];
	view[ 1 ] = 0.0;
 
	// For even-length sequences, zero out the imaginary part of the Nyquist component:
	if ( isEven( N ) ) {
		view[ ( 2*M ) - 1 ] = 0.0;
	}
}
 
/**
* Computes the forward FFT of a real-valued input.
*
* @private
* @param {ArrayLikeObject<Object>} arrays - array-like object containing ndarrays
* @returns {void}
*/
function gkernel( arrays ) {
	kernel( reinterpret, grffti, grfftf, Float64Array, arrays );
}

/**
* Computes the one-dimensional discrete Fourier transform of a real-valued input ndarray and assigns the results to a provided output ndarray.
*
* @private
* @name rfftf
* @type {Function}
* @param {ndarray} out - output ndarray
* @param {ndarray} x - input ndarray
* @throws {TypeError} first argument must be an ndarray-like object
* @throws {TypeError} second argument must be an ndarray-like object
* @returns {ndarray} output ndarray
*
* @example
* var Float64Array = require( '@stdlib/array/float64' );
* var Complex128Array = require( '@stdlib/array/complex128' );
* var ndarray = require( '@stdlib/ndarray/ctor' );
*
* // Create a data buffer:
* var xbuf = new Float64Array( [ 1.0, 2.0, 3.0, 4.0 ] );
*
* // Define the shape of the input array:
* var sh = [ 4 ];
*
* // Define the array strides:
* var sx = [ 1 ];
*
* // Define the index offset:
* var ox = 0;
*
* // Create an input ndarray:
* var x = new ndarray( 'float64', xbuf, sh, sx, ox, 'row-major' );
*
* // Create an output ndarray:
* var out = new ndarray( 'complex128', new Complex128Array( 3 ), [ 3 ], [ 1 ], 0, 'row-major' );
*
* // Perform operation:
* var v = rfftf( out, x );
* // returns <ndarray>[ <Complex128>[ 10.0, 0.0 ], <Complex128>[ -2.0, 2.0 ], <Complex128>[ -2.0, 0.0 ] ]
*/
var rfftf = factory( table, [ idtypes ], odtypes );
