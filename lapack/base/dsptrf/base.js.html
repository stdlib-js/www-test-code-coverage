// Factor used to determine the pivot block size (see LAPACK reference DSPTRF):
var ALPHA = ( 1.0 + sqrt( 17.0 ) ) / 8.0;

/**
* Resolves the physical index of a (one-based) packed element.
*
* @private
* @param {NonNegativeInteger} oa - starting index for `AP`
* @param {integer} sa - stride length for `AP`
* @param {PositiveInteger} idx - one-based packed index
* @returns {integer} physical index into `AP`
*/
function pos( oa, sa, idx ) {
	return oa + ( ( idx-1 ) * sa );
}
 
/**
* Factorizes a real symmetric matrix `A = U*D*U^T` using the upper triangle of `A` supplied in column-major packed form.
*
* @private
* @param {NonNegativeInteger} N - order of the matrix `A`
* @param {Float64Array} AP - packed form of a symmetric matrix `A`
* @param {integer} sa - stride length for `AP`
* @param {NonNegativeInteger} oa - starting index for `AP`
* @param {Int32Array} IPIV - vector of pivot indices
* @param {integer} si - stride length for `IPIV`
* @param {NonNegativeInteger} oi - starting index for `IPIV`
* @returns {integer} status code
*/
function factorU( N, AP, sa, oa, IPIV, si, oi ) {
	var absakk;
	var colmax;
	var rowmax;
	var kstep;
	var ijm1;
	var wkm1;
	var ckm1;
	var info;
	var imax;
	var jmax;
	var d11;
	var d12;
	var d22;
	var knc;
	var kpc;
	var ck;
	var cj;
	var i1;
	var i2;
	var ii;
	var ij;
	var r1;
	var wk;
	var kc;
	var kk;
	var kp;
	var kx;
	var i;
	var j;
	var k;
	var t;
 
	info = 0;
 
	// `k` is the main loop index, decreasing from `N` to `1` in steps of `1` or `2`:
	k = N;
	kc = ( ( N-1 )*N / 2 ) + 1;
	while ( k >= 1 ) {
		knc = kc;
		kstep = 1;
 
		// Determine rows and columns to be interchanged and whether a 1-by-1 or 2-by-2 pivot block will be used...
		absakk = abs( AP[ pos( oa, sa, kc+k-1 ) ] );
 
		// `imax` is the row-index of the largest off-diagonal element in column `k`, and `colmax` is its absolute value...
		if ( k > 1 ) {
			imax = idamax( k-1, AP, sa, pos( oa, sa, kc ) ) + 1;
			colmax = abs( AP[ pos( oa, sa, kc+imax-1 ) ] );
		} else {
			colmax = 0.0;
		}
		if ( max( absakk, colmax ) === 0.0 ) {
			// Column `k` is zero: set `info` and continue...
			if ( info === 0 ) {
				info = k;
			}
			kp = k;
		} else {
			if ( absakk >= ALPHA*colmax ) {
				// No interchange, use 1-by-1 pivot block...
				kp = k;
			} else {
				// `jmax` is the column-index of the largest off-diagonal element in row `imax`, and `rowmax` is its absolute value...
				rowmax = 0.0;
				jmax = imax;
				kx = ( imax*( imax+1 ) / 2 ) + imax;
				for ( j = imax+1; j <= k; j++ ) {
					t = abs( AP[ pos( oa, sa, kx ) ] );
					if ( t > rowmax ) {
						rowmax = t;
						jmax = j;
					}
					kx += j;
				}
				kpc = ( ( imax-1 )*imax / 2 ) + 1;
				if ( imax > 1 ) {
					jmax = idamax( imax-1, AP, sa, pos( oa, sa, kpc ) ) + 1;
					rowmax = max( rowmax, abs( AP[ pos( oa, sa, kpc+jmax-1 ) ] ) );
				}
				if ( absakk >= ALPHA*colmax*( colmax / rowmax ) ) {
					// No interchange, use 1-by-1 pivot block...
					kp = k;
				} else if ( abs( AP[ pos( oa, sa, kpc+imax-1 ) ] ) >= ALPHA*rowmax ) {
					// Interchange rows and columns `k` and `imax`, use 1-by-1 pivot block...
					kp = imax;
				} else {
					// Interchange rows and columns `k-1` and `imax`, use 2-by-2 pivot block...
					kp = imax;
					kstep = 2;
				}
			}
			kk = k - kstep + 1;
			if ( kstep === 2 ) {
				knc = knc - k + 1;
			}
			if ( kp !== kk ) {
				// Interchange rows and columns `kk` and `kp` in the leading submatrix `A(0:k-1,0:k-1)` (zero-based)...
				dswap( kp-1, AP, sa, pos( oa, sa, knc ), AP, sa, pos( oa, sa, kpc ) );
				kx = kpc + kp - 1;
				for ( j = kp+1; j <= kk-1; j++ ) {
					kx = kx + j - 1;
					i1 = pos( oa, sa, knc+j-1 );
					i2 = pos( oa, sa, kx );
					t = AP[ i1 ];
					AP[ i1 ] = AP[ i2 ];
					AP[ i2 ] = t;
				}
				i1 = pos( oa, sa, knc+kk-1 );
				i2 = pos( oa, sa, kpc+kp-1 );
				t = AP[ i1 ];
				AP[ i1 ] = AP[ i2 ];
				AP[ i2 ] = t;
				if ( kstep === 2 ) {
					i1 = pos( oa, sa, kc+k-2 );
					i2 = pos( oa, sa, kc+kp-1 );
					t = AP[ i1 ];
					AP[ i1 ] = AP[ i2 ];
					AP[ i2 ] = t;
				}
			}
			// Update the leading submatrix...
			if ( kstep === 1 ) {
				// 1-by-1 pivot block `D(k)`: column `k` now holds `W(k) = U(k)*D(k)`, where `U(k)` is the k-th column of `U`. Perform a rank-1 update of `A(0:k-2,0:k-2)` (zero-based) as `A := A - U(k)*D(k)*U(k)^T = A - W(k)*(1/D(k))*W(k)^T` and store `U(k)` in column `k`...
				r1 = 1.0 / AP[ pos( oa, sa, kc+k-1 ) ];
				dspr( 'column-major', 'upper', k-1, -r1, AP, sa, pos( oa, sa, kc ), AP, sa, oa );
				dscal( k-1, r1, AP, sa, pos( oa, sa, kc ) );
			} else if ( k > 2 ) {
				// 2-by-2 pivot block `D(k)`. Perform a rank-2 update of `A(0:k-3,0:k-3)` (zero-based)...
 
				// Resolve the packed base offsets for columns `k` and `k-1`...
				ck = ( ( k-1 )*k ) / 2;
				ckm1 = ( ( k-2 )*( k-1 ) ) / 2;
				d12 = AP[ pos( oa, sa, k-1 + ck ) ];
				d22 = AP[ pos( oa, sa, k-1 + ckm1 ) ] / d12;
				d11 = AP[ pos( oa, sa, k + ck ) ] / d12;
				t = 1.0 / ( ( d11*d22 ) - 1.0 );
				d12 = t / d12;
				for ( j = k-2; j >= 1; j-- ) {
					cj = ( ( j-1 )*j ) / 2;
					ij = pos( oa, sa, j + ck );
					ijm1 = pos( oa, sa, j + ckm1 );
					wkm1 = d12 * ( ( d11*AP[ ijm1 ] ) - AP[ ij ] );
					wk = d12 * ( ( d22*AP[ ij ] ) - AP[ ijm1 ] );
					for ( i = j; i >= 1; i-- ) {
						ii = pos( oa, sa, i + cj );
						AP[ ii ] = AP[ ii ] - ( AP[ pos( oa, sa, i + ck ) ]*wk ) - ( AP[ pos( oa, sa, i + ckm1 ) ]*wkm1 );
					}
					AP[ ij ] = wk;
					AP[ ijm1 ] = wkm1;
				}
			}
		}
		// Store details of the interchanges in `IPIV`...
		if ( kstep === 1 ) {
			IPIV[ oi + ( ( k-1 )*si ) ] = kp;
		} else {
			IPIV[ oi + ( ( k-1 )*si ) ] = -kp;
			IPIV[ oi + ( ( k-2 )*si ) ] = -kp;
		}
		// Decrease `k` and return to the start of the main loop...
		k -= kstep;
		kc = knc - k;
	}
	return info;
}
 
/**
* Factorizes a real symmetric matrix `A = L*D*L^T` using the lower triangle of `A` supplied in column-major packed form.
*
* @private
* @param {NonNegativeInteger} N - order of the matrix `A`
* @param {Float64Array} AP - packed form of a symmetric matrix `A`
* @param {integer} sa - stride length for `AP`
* @param {NonNegativeInteger} oa - starting index for `AP`
* @param {Int32Array} IPIV - vector of pivot indices
* @param {integer} si - stride length for `IPIV`
* @param {NonNegativeInteger} oi - starting index for `IPIV`
* @returns {integer} status code
*/
function factorL( N, AP, sa, oa, IPIV, si, oi ) {
	var absakk;
	var colmax;
	var rowmax;
	var kstep;
	var ijp1;
	var wkp1;
	var ckp1;
	var info;
	var imax;
	var jmax;
	var npp;
	var d11;
	var d21;
	var d22;
	var knc;
	var kpc;
	var ck;
	var cj;
	var i1;
	var i2;
	var ii;
	var ij;
	var r1;
	var wk;
	var kc;
	var kk;
	var kp;
	var kx;
	var i;
	var j;
	var k;
	var t;
 
	info = 0;
 
	// `k` is the main loop index, increasing from `1` to `N` in steps of `1` or `2`:
	k = 1;
	kc = 1;
	npp = N*( N+1 ) / 2;
	while ( k <= N ) {
		knc = kc;
		kstep = 1;
 
		// Determine rows and columns to be interchanged and whether a 1-by-1 or 2-by-2 pivot block will be used...
		absakk = abs( AP[ pos( oa, sa, kc ) ] );
 
		// `imax` is the row-index of the largest off-diagonal element in column `k`, and `colmax` is its absolute value...
		if ( k < N ) {
			imax = k + idamax( N-k, AP, sa, pos( oa, sa, kc+1 ) ) + 1;
			colmax = abs( AP[ pos( oa, sa, kc+imax-k ) ] );
		} else {
			colmax = 0.0;
		}
		if ( max( absakk, colmax ) === 0.0 ) {
			// Column `k` is zero: set `info` and continue...
			if ( info === 0 ) {
				info = k;
			}
			kp = k;
		} else {
			if ( absakk >= ALPHA*colmax ) {
				// No interchange, use 1-by-1 pivot block...
				kp = k;
			} else {
				// `jmax` is the column-index of the largest off-diagonal element in row `imax`, and `rowmax` is its absolute value...
				rowmax = 0.0;
				kx = kc + imax - k;
				for ( j = k; j <= imax-1; j++ ) {
					t = abs( AP[ pos( oa, sa, kx ) ] );
					if ( t > rowmax ) {
						rowmax = t;
						jmax = j;
					}
					kx = kx + N - j;
				}
				kpc = npp - ( ( N-imax+1 )*( N-imax+2 ) / 2 ) + 1;
				if ( imax < N ) {
					jmax = imax + idamax( N-imax, AP, sa, pos( oa, sa, kpc+1 ) ) + 1;
					rowmax = max( rowmax, abs( AP[ pos( oa, sa, kpc+jmax-imax ) ] ) );
				}
				if ( absakk >= ALPHA*colmax*( colmax / rowmax ) ) {
					// No interchange, use 1-by-1 pivot block...
					kp = k;
				} else if ( abs( AP[ pos( oa, sa, kpc ) ] ) >= ALPHA*rowmax ) {
					// Interchange rows and columns `k` and `imax`, use 1-by-1 pivot block...
					kp = imax;
				} else {
					// Interchange rows and columns `k+1` and `imax`, use 2-by-2 pivot block...
					kp = imax;
					kstep = 2;
				}
			}
			kk = k + kstep - 1;
			if ( kstep === 2 ) {
				knc = knc + N - k + 1;
			}
			if ( kp !== kk ) {
				// Interchange rows and columns `kk` and `kp` in the trailing submatrix `A(k-1:n-1,k-1:n-1)` (zero-based)...
				if ( kp < N ) {
					dswap( N-kp, AP, sa, pos( oa, sa, knc+kp-kk+1 ), AP, sa, pos( oa, sa, kpc+1 ) );
				}
				kx = knc + kp - kk;
				for ( j = kk+1; j <= kp-1; j++ ) {
					kx = kx + N - j + 1;
					i1 = pos( oa, sa, knc+j-kk );
					i2 = pos( oa, sa, kx );
					t = AP[ i1 ];
					AP[ i1 ] = AP[ i2 ];
					AP[ i2 ] = t;
				}
				i1 = pos( oa, sa, knc );
				i2 = pos( oa, sa, kpc );
				t = AP[ i1 ];
				AP[ i1 ] = AP[ i2 ];
				AP[ i2 ] = t;
				if ( kstep === 2 ) {
					i1 = pos( oa, sa, kc+1 );
					i2 = pos( oa, sa, kc+kp-k );
					t = AP[ i1 ];
					AP[ i1 ] = AP[ i2 ];
					AP[ i2 ] = t;
				}
			}
			// Update the trailing submatrix...
			if ( kstep === 1 ) {
				// 1-by-1 pivot block `D(k)`: column `k` now holds `W(k) = L(k)*D(k)`, where `L(k)` is the k-th column of `L`. Perform a rank-1 update of `A(k:n-1,k:n-1)` (zero-based) as `A := A - L(k)*D(k)*L(k)^T = A - W(k)*(1/D(k))*W(k)^T` and store `L(k)` in column `k`...
				if ( k < N ) {
					r1 = 1.0 / AP[ pos( oa, sa, kc ) ];
					dspr( 'column-major', 'lower', N-k, -r1, AP, sa, pos( oa, sa, kc+1 ), AP, sa, pos( oa, sa, kc+N-k+1 ) );
					dscal( N-k, r1, AP, sa, pos( oa, sa, kc+1 ) );
				}
			} else if ( k < N-1 ) {
				// 2-by-2 pivot block `D(k)`. Perform a rank-2 update of `A(k+1:n-1,k+1:n-1)` (zero-based)...
 
				// Resolve the packed base offsets for columns `k` and `k+1`...
				ck = ( ( k-1 )*( ( 2*N )-k ) ) / 2;
				ckp1 = ( k*( ( 2*N )-k-1 ) ) / 2;
				d21 = AP[ pos( oa, sa, k+1 + ck ) ];
				d11 = AP[ pos( oa, sa, k+1 + ckp1 ) ] / d21;
				d22 = AP[ pos( oa, sa, k + ck ) ] / d21;
				t = 1.0 / ( ( d11*d22 ) - 1.0 );
				d21 = t / d21;
				for ( j = k+2; j <= N; j++ ) {
					cj = ( ( j-1 )*( ( 2*N )-j ) ) / 2;
					ij = pos( oa, sa, j + ck );
					ijp1 = pos( oa, sa, j + ckp1 );
					wk = d21 * ( ( d11*AP[ ij ] ) - AP[ ijp1 ] );
					wkp1 = d21 * ( ( d22*AP[ ijp1 ] ) - AP[ ij ] );
					for ( i = j; i <= N; i++ ) {
						ii = pos( oa, sa, i + cj );
						AP[ ii ] = AP[ ii ] - ( AP[ pos( oa, sa, i + ck ) ]*wk ) - ( AP[ pos( oa, sa, i + ckp1 ) ]*wkp1 );
					}
					AP[ ij ] = wk;
					AP[ ijp1 ] = wkp1;
				}
			}
		}
		// Store details of the interchanges in `IPIV`...
		if ( kstep === 1 ) {
			IPIV[ oi + ( ( k-1 )*si ) ] = kp;
		} else {
			IPIV[ oi + ( ( k-1 )*si ) ] = -kp;
			IPIV[ oi + ( k*si ) ] = -kp;
		}
		// Increase `k` and return to the start of the main loop...
		k += kstep;
		kc = knc + N - k + 2;
	}
	return info;
}

/**
* Computes the factorization of a real symmetric matrix `A` stored in packed format using the Bunch-Kaufman diagonal pivoting method.
*
* @private
* @param {string} order - storage layout
* @param {string} uplo - specifies whether the upper or lower triangular part of the symmetric matrix `A` is supplied
* @param {NonNegativeInteger} N - order of the matrix `A`
* @param {Float64Array} AP - packed form of a symmetric matrix `A`
* @param {integer} strideAP - stride length for `AP`
* @param {NonNegativeInteger} offsetAP - starting index for `AP`
* @param {Int32Array} IPIV - vector of pivot indices
* @param {integer} strideIPIV - stride length for `IPIV`
* @param {NonNegativeInteger} offsetIPIV - starting index for `IPIV`
* @returns {integer} status code
*
* @example
* var Float64Array = require( '@stdlib/array/float64' );
* var Int32Array = require( '@stdlib/array/int32' );
*
* var AP = new Float64Array( [ 4.0, 1.0, 2.0, -2.0, 0.0, 3.0, 2.0, 1.0, -2.0, -1.0 ] );
* var IPIV = new Int32Array( 4 );
*
* dsptrf( 'column-major', 'upper', 4, AP, 1, 0, IPIV, 1, 0 );
* // IPIV => <Int32Array>[ 1, 2, 3, 1 ]
*/
function dsptrf( order, uplo, N, AP, strideAP, offsetAP, IPIV, strideIPIV, offsetIPIV ) {
	if ( N === 0 ) {
		return 0;
	}
	// For a symmetric matrix, an upper triangle stored in row-major order is identical in memory to a lower triangle stored in column-major order (and vice versa), so resolve `order`+`uplo` to the equivalent column-major factorization branch...
	if (
		( isColumnMajor( order ) && uplo === 'upper' ) ||
		( isRowMajor( order ) && uplo === 'lower' )
	) {
		return factorU( N, AP, strideAP, offsetAP, IPIV, strideIPIV, offsetIPIV );
	}
	return factorL( N, AP, strideAP, offsetAP, IPIV, strideIPIV, offsetIPIV );
}
